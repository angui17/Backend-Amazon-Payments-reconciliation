using System;
using System.Data.Odbc;
using Newtonsoft.Json.Linq;
using System.Collections.Generic;

string resultJson = "";

// Safe getters (JObject o IDictionary)
string GetStr(string key)
{
    if (InputJson == null) return null;

    var jo = InputJson as JObject;
    if (jo != null)
    {
        JToken t;
        if (!jo.TryGetValue(key, out t) || t == null || t.Type == JTokenType.Null) return null;
        var s = t.ToString();
        if (string.Equals(s, "null", StringComparison.OrdinalIgnoreCase)) return null;
        return s;
    }

    var dict = InputJson as IDictionary<string, object>;
    if (dict != null)
    {
        if (!dict.ContainsKey(key)) return null;
        var val = dict[key];
        if (val == null || val is DBNull) return null;
        var s = Convert.ToString(val);
        if (string.IsNullOrWhiteSpace(s) || string.Equals(s, "null", StringComparison.OrdinalIgnoreCase)) return null;
        return s;
    }

    return null;
}

int? GetInt(string key)
{
    if (InputJson == null) return null;

    var jo = InputJson as JObject;
    if (jo != null)
    {
        JToken t;
        if (!jo.TryGetValue(key, out t) || t == null || t.Type == JTokenType.Null) return null;
        try { return t.ToObject<int?>(); } catch { return null; }
    }

    var dict = InputJson as IDictionary<string, object>;
    if (dict != null)
    {
        if (!dict.ContainsKey(key)) return null;
        var val = dict[key];
        if (val == null || val is DBNull) return null;
        try { return Convert.ToInt32(val); } catch { return null; }
    }

    return null;
}

// Inputs
string fecha_desde = GetStr("fecha_desde");
string fecha_hasta = GetStr("fecha_hasta");
string status      = GetStr("status");
int? limitRecords  = GetInt("limit_records");

// Defaults
if (string.IsNullOrWhiteSpace(fecha_desde)) fecha_desde = "01-01-2000";
if (string.IsNullOrWhiteSpace(fecha_hasta)) fecha_hasta = DateTime.Now.ToString("MM-dd-yyyy");

if (string.IsNullOrWhiteSpace(status)) status = "ALL";
status = status.Trim().ToUpper();

int limit_records = (limitRecords.HasValue && limitRecords.Value > 0) ? limitRecords.Value : 200;

using (var connection = new OdbcConnection("{ConnectionString}"))
{
    connection.Open();

    using (var command = new OdbcCommand("{ CALL \"CTS_ACCOUNTING_POSTING_STATUS\"( ?, ?, ?, ? ) }", connection))
    {
        command.Parameters.Add("fecha_desde", OdbcType.VarChar).Value = fecha_desde;
        command.Parameters.Add("fecha_hasta", OdbcType.VarChar).Value = fecha_hasta;
        command.Parameters.Add("p_status", OdbcType.VarChar).Value = status;
        command.Parameters.Add("limit_records", OdbcType.Int).Value = limit_records;

        using (var reader = command.ExecuteReader())
        {
            if (reader.Read() && !reader.IsDBNull(0))
                resultJson = reader.GetString(0);
        }
    }
}

if (string.IsNullOrWhiteSpace(resultJson))
{
    return new JObject
    {
        ["id"] = -500,
        ["Message"] = "SP no devolvió datos (JSONRESULT NULL/empty)."
    };
}

try { return JObject.Parse(resultJson); }
catch
{
    return new JObject
    {
        ["id"] = -501,
        ["Message"] = "SP devolvió un JSON inválido."
    };
}
