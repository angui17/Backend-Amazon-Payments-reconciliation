using System;
using System.Data.Odbc;
using Newtonsoft.Json.Linq;

string resultJson = "";

// Required
string settlementId = InputJson["settlementId"]?.ToString();

// Optional
int? limitRows = InputJson["limit_rows"]?.ToObject<int?>();
string txnTypesCsv = InputJson["txn_types_csv"]?.ToString();
string amountDescLike = InputJson["amount_desc_like"]?.ToString();

// Validate
if (string.IsNullOrWhiteSpace(settlementId))
{
    return new JObject
    {
        ["id"] = -400,
        ["Message"] = "Parameter 'settlementId' is required. Example: { \"settlementId\": \"23405515541\", \"limit_rows\": 5000 }"
    };
}
settlementId = settlementId.Trim();

// Defaults
int limit_rows = (limitRows.HasValue && limitRows.Value > 0) ? limitRows.Value : 5000;
txnTypesCsv = (txnTypesCsv ?? "").Trim();
amountDescLike = (amountDescLike ?? "").Trim();

using (var connection = new OdbcConnection("{ConnectionString}"))
{
    connection.Open();

    using (var command = new OdbcCommand("{ CALL \"CTS_REPORT_SETTLEMENT_EXPORT\"( ?, ?, ?, ? ) }", connection))
    {
        // SP signature: (p_settlement_id, limit_rows, p_txn_types_csv, p_amount_desc_like)
        command.Parameters.Add("p_settlement_id", OdbcType.VarChar).Value = settlementId;
        command.Parameters.Add("limit_rows", OdbcType.Int).Value = limit_rows;

        // Pass NULL when empty (so SP treats as NULL/optional)
        if (string.IsNullOrWhiteSpace(txnTypesCsv))
            command.Parameters.Add("p_txn_types_csv", OdbcType.VarChar).Value = DBNull.Value;
        else
            command.Parameters.Add("p_txn_types_csv", OdbcType.VarChar).Value = txnTypesCsv;

        if (string.IsNullOrWhiteSpace(amountDescLike))
            command.Parameters.Add("p_amount_desc_like", OdbcType.VarChar).Value = DBNull.Value;
        else
            command.Parameters.Add("p_amount_desc_like", OdbcType.VarChar).Value = amountDescLike;

        using (var reader = command.ExecuteReader())
        {
            if (reader.Read())
            {
                resultJson = reader.GetString(0);
            }
        }
    }
}

if (string.IsNullOrWhiteSpace(resultJson))
{
    return new JObject
    {
        ["id"] = -500,
        ["Message"] = "SP no devolvió datos."
    };
}

try
{
    return JObject.Parse(resultJson);
}
catch
{
    return new JObject
    {
        ["id"] = -501,
        ["Message"] = "SP devolvió un JSON inválido."
    };
}
