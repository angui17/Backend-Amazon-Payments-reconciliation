// Método principal que procesa un archivo CSV
 async Task<string> ProcessFileAsync(string filePath){
    var fileName = Path.GetFileName(filePath);
    //log.LogInformation($"Processing CSV file: {fileName}");
    try    {
        // Leer el contenido del archivo
        var fileContent = await File.ReadAllTextAsync(filePath);
        // Procesar el contenido CSV
        string result = await ProcessCsvFile(fileContent, fileName);
        return JsonSerializer.Serialize(new {success = true, message = result});
        // Mover archivo a carpeta de procesados
        MoveToProcessedFolder(filePath, fileName);
        //log.LogInformation($"Successfully processed CSV file: {fileName}");
    }
    catch (Exception ex)    {
        //log.LogError(ex, $"Error processing CSV file: {fileName}");
        return JsonSerializer.Serialize(new {success = false, message = ex.Message});
        MoveToFailedFolder(filePath, fileName);
        throw;
    }
}

// Convierte el string CSV en un DataTable
 DataTable ReadCsvToDataTableFromString(string csvContent){
    DataTable dt = new DataTable();
    using (StringReader sr = new StringReader(csvContent))
    {
        // Leer headers y limpiar nombres de columnas
        string[] headers = sr.ReadLine().Split('|');
        foreach (string header in headers)
        {
            dt.Columns.Add(header.Replace("\"","").Replace("/", "").Replace(" ", "").ToUpper());
        }
        // Leer y procesar cada línea de datos
        while (sr.Peek() != -1)        {
            string line = sr.ReadLine();
            if (!string.IsNullOrEmpty(line))            {
                string[] rows = line.Split('|');
                DataRow dr = dt.NewRow();
                for (int i = 0; i < headers.Length && i < rows.Length; i++)                {
                    dr[i] = rows[i].Replace("\"", "").ToUpper();
                }
                dt.Rows.Add(dr);
            }
        }
    }
    return dt;
}

    command.Parameters.AddWithValue("@ORDER_POSTAL", parameters.ORDER_POSTAL);
    command.Parameters.AddWithValue("@TAX_COLLECTION_MODEL", parameters.TAX_COLLECTION_MODEL);
    command.Parameters.AddWithValue("@PRODUCT_SALES", parameters.PRODUCT_SALES);
    command.Parameters.AddWithValue("@PRODUCT_SALES_TAX", parameters.PRODUCT_SALES_TAX);
    command.Parameters.AddWithValue("@SHIPPING_CREDITS", parameters.SHIPPING_CREDITS);

// Mueve el archivo a la carpeta de procesados
 void MoveToProcessedFolder(string originalFilePath, string fileName){
    var destinationPath = Path.Combine(ProcessedFolder, fileName);
    File.Move(originalFilePath, destinationPath, true);
}

// Mueve el archivo a la carpeta de fallidos
 void MoveToFailedFolder(string originalFilePath, string fileName){
    var destinationPath = Path.Combine(FailedFolder, fileName);
    File.Move(originalFilePath, destinationPath, true);
}
