using Microsoft.Azure.Functions.Worker;
using Microsoft.Extensions.Logging;
using Azure.Storage.Blobs;

    public class CsvProcessingFunction
    {
        private readonly ILogger<CsvProcessingFunction> _logger;
        private readonly BlobServiceClient _blobServiceClient;
        private readonly string _sqlConnectionString;

        public CsvProcessingFunction(ILogger<CsvProcessingFunction> logger)
        {
            _logger = logger;
            
            var storageConnectionString = Environment.GetEnvironmentVariable("AzureWebJobsStorage");
            _sqlConnectionString = Environment.GetEnvironmentVariable("SqlConnectionString");
            
            _blobServiceClient = new BlobServiceClient(storageConnectionString);
        }

        [Function("CsvFileProcessor")]
        public async Task Run(
            [BlobTrigger("pending-csv/{name}")] BlobClient blobClient,
            string name,
            FunctionContext context)
        {
            _logger.LogInformation($"Processing CSV file: {name}");

            try
            {
                // Descargar el archivo CSV
                var blobContent = await DownloadBlobContent(blobClient);
                
                // Procesar el contenido CSV (igual que tu código original)
                await ProcessCsvFile(blobContent, name);
                
                // Mover archivo a contenedor de procesados
                await MoveToProcessedContainer(blobClient, name);
                
                _logger.LogInformation($"Successfully processed CSV file: {name}");
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, $"Error processing CSV file: {name}");
                await MoveToFailedContainer(blobClient, name);
                throw;
            }
        }

        private async Task<string> DownloadBlobContent(BlobClient blobClient)
        {
            try
            {
                BlobDownloadResult downloadResult = await blobClient.DownloadContentAsync();
                return downloadResult.Content.ToString();
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error downloading blob content");
                throw;
            }
        }

        private DataTable ReadCsvToDataTableFromString(string csvContent)
        {
            DataTable dt = new DataTable();
            using (StringReader sr = new StringReader(csvContent))
            {
                string[] headers = sr.ReadLine().Split('|');
                foreach (string header in headers)
                {
                    dt.Columns.Add(header.Replace("\"","").Replace("/", "_").Replace(" ", "_").ToUpper());
                }

                while (sr.Peek() != -1)
                {
                    string line = sr.ReadLine();
                    if (!string.IsNullOrEmpty(line))
                    {
                        string[] rows = line.Split('|');
                        DataRow dr = dt.NewRow();
                        for (int i = 0; i < headers.Length && i < rows.Length; i++)
                        {
                            dr[i] = rows[i].Replace("\"", "").ToUpper();
                        }
                        dt.Rows.Add(dr);
                    }
                }
            }
            return dt;
        }

        private async Task MoveToProcessedContainer(BlobClient originalBlob, string fileName)
        {
            var processedContainer = _blobServiceClient.GetBlobContainerClient("processed-csv");
            await processedContainer.CreateIfNotExistsAsync();

            var processedBlob = processedContainer.GetBlobClient(fileName);
            await processedBlob.StartCopyFromUriAsync(originalBlob.Uri);
            await originalBlob.DeleteIfExistsAsync();
        }

        private async Task MoveToFailedContainer(BlobClient originalBlob, string fileName)
        {
            var failedContainer = _blobServiceClient.GetBlobContainerClient("failed-csv");
            await failedContainer.CreateIfNotExistsAsync();

            var failedBlob = failedContainer.GetBlobClient(fileName);
            await failedBlob.StartCopyFromUriAsync(originalBlob.Uri);
            await originalBlob.DeleteIfExistsAsync();
        }
    }

    // API Controller para monitoreo (mantener igual)
    [ApiController]
    [Route("api/[controller]")]
    public class FileProcessingController : ControllerBase
    {
        private readonly ILogger<FileProcessingController> _logger;
        private readonly BlobServiceClient _blobServiceClient;

        public FileProcessingController(ILogger<FileProcessingController> logger)
        {
            _logger = logger;
            var storageConnectionString = Environment.GetEnvironmentVariable("AzureWebJobsStorage");
            _blobServiceClient = new BlobServiceClient(storageConnectionString);
        }

        [HttpGet("status")]
        public async Task<IActionResult> GetProcessingStatus()
        {
            var status = new
            {
                PendingFiles = await GetContainerFileCount("pending-csv"),
                ProcessedFiles = await GetContainerFileCount("processed-csv"),
                FailedFiles = await GetContainerFileCount("failed-csv"),
                LastProcessed = DateTime.UtcNow.AddHours(-1)
            };

            return Ok(status);
        }

        [HttpGet("status/{fileName}")]
        public async Task<IActionResult> GetFileStatus(string fileName)
        {
            var containers = new[] { "pending-csv", "processed-csv", "failed-csv" };

            foreach (var containerName in containers)
            {
                var container = _blobServiceClient.GetBlobContainerClient(containerName);
                if (await container.ExistsAsync())
                {
                    var blob = container.GetBlobClient(fileName);
                    if (await blob.ExistsAsync())
                    {
                        var properties = await blob.GetPropertiesAsync();
                        return Ok(new
                        {
                            FileName = fileName,
                            Status = containerName.Replace("-csv", "").ToUpper(),
                            LastModified = properties.Value.LastModified,
                            Size = properties.Value.ContentLength
                        });
                    }
                }
            }

            return NotFound($"File {fileName} not found in any container");
        }

        private async Task<int> GetContainerFileCount(string containerName)
        {
            var container = _blobServiceClient.GetBlobContainerClient(containerName);
            if (!await container.ExistsAsync())
                return 0;

            var count = 0;
            await foreach (var blobItem in container.GetBlobsAsync())
            {
                count++;
            }
            return count;
        }
