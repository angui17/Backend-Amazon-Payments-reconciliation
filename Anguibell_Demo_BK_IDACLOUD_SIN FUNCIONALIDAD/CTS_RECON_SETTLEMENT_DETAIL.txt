using System;
using System.Data.Odbc;
using Newtonsoft.Json.Linq;

string resultJson = "";

string settlementId = InputJson["settlementId"]?.ToString();
int? limit = InputJson["limit"]?.ToObject<int?>();

int limitRows = (limit.HasValue && limit.Value > 0) ? limit.Value : 500;

if (string.IsNullOrWhiteSpace(settlementId))
{
    return new JObject
    {
        ["id"] = -400,
        ["Message"] = "Parameter 'settlementId' is required. Example: { \"settlementId\": \"23405515541\" }"
    };
}

settlementId = settlementId.Trim();

using (var connection = new OdbcConnection("{ConnectionString}"))
{
    connection.Open();

    using (var command = new OdbcCommand("{ CALL \"CTS_RECON_SETTLEMENT_DETAIL\"( ?, ? ) }", connection))
    {
        command.Parameters.Add("p_settlement_id", OdbcType.VarChar).Value = settlementId;
        command.Parameters.Add("limit_rows", OdbcType.Int).Value = limitRows;

        using (var reader = command.ExecuteReader())
        {
            if (reader.Read())
            {
                resultJson = reader.GetString(0);
            }
        }
    }
}

if (string.IsNullOrWhiteSpace(resultJson))
{
    return new JObject
    {
        ["id"] = -500,
        ["Message"] = "SP no devolvió datos."
    };
}

try
{
    return JToken.Parse(resultJson);
}
catch
{
    return new JObject
    {
        ["id"] = -501,
        ["Message"] = "SP devolvió un JSON inválido."
    };
}
